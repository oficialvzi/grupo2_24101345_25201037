# Etapa 3 - Parte B: backup e restauração

$ mysql -u root --default-character-set=utf8mb4 locadora -e "DROP TABLE IF EXISTS teste_restore; CREATE TABLE teste_restore (id INT PRIMARY KEY AUTO_INCREMENT, descricao VARCHAR(100), criado_em DATETIME DEFAULT CURRENT_TIMESTAMP); INSERT INTO teste_restore (descricao) VALUES ('registro 1 antes do backup'), ('registro 2 antes do backup');" && echo OK
OK

$ mysql -u root --default-character-set=utf8mb4 -t locadora -e 'SELECT * FROM teste_restore;'
+----+----------------------------+---------------------+
| id | descricao                  | criado_em           |
+----+----------------------------+---------------------+
|  1 | registro 1 antes do backup | 2026-11-04 10:43:46 |
|  2 | registro 2 antes do backup | 2026-11-04 10:43:46 |
+----+----------------------------+---------------------+

$ mysqldump -u root --default-character-set=utf8mb4 --routines --triggers --databases locadora > locadora/backup/backup_locadora.sql && echo OK
OK

$ ls -lh locadora/backup/backup_locadora.sql | awk '{print $5, $9}'
21K locadora/backup/backup_locadora.sql
  tabelas no backup:  7
  triggers no backup: 3
  rotinas no backup:  2

# Simulando a perda de dados

$ mysql -u root --default-character-set=utf8mb4 locadora -e 'DROP TABLE teste_restore;' && echo OK
OK

$ mysql -u root --default-character-set=utf8mb4 -t locadora -e 'SELECT * FROM teste_restore;' 2>&1
ERROR 1146 (42S02) at line 1: Table 'locadora.teste_restore' doesn't exist

# Restaurando a partir do backup

$ mysql -u root --default-character-set=utf8mb4 < locadora/backup/backup_locadora.sql && echo OK
OK

$ mysql -u root --default-character-set=utf8mb4 -t locadora -e 'SELECT * FROM teste_restore;'
